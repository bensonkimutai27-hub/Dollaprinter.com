/** * Dollarprinter Clone - Deriv API Integration * Core Logic for WebSocket Connection, Authentication, and Live Data Streaming */ class DerivTradingEngine { constructor(appId) { // Deriv WebSocket endpoint (using official App ID) this.wsUrl = `wss://://derivws.com{appId}`; this.socket = null; this.isAuthorized = false; } // 1. Initialize the WebSocket Connection init() { this.socket = new WebSocket(this.wsUrl); this.socket.onopen = (connection) => { console.log("Successfully connected to Deriv Exchange API Server."); // Automatically ping every 30 seconds to keep the connection alive this.startKeepAlive(); }; this.socket.onmessage = (message) => { const data = JSON.parse(message.data); this.handleIncomingData(data); }; this.socket.onerror = (error) => { console.error("WebSocket Error: ", error); }; this.socket.onclose = () => { console.log("Connection closed. Reconnecting..."); setTimeout(() => this.init(), 5000); // Auto-reconnect after 5 seconds }; } // 2. Authenticate the User using their API Token authorize(apiToken) { if (this.socket.readyState === WebSocket.OPEN) { this.socket.send(JSON.stringify({ authorize: apiToken })); } else { console.error("Cannot authorize. Socket is not open."); } } // 3. Subscribe to Live Market Assets (e.g., Volatility 100 Index or Forex pairs) subscribeToTicks(symbol = 'R_100') { if (this.socket.readyState === WebSocket.OPEN) { console.log(`Subscribing to real-time ticks for: ${symbol}`); this.socket.send(JSON.stringify({ ticks: symbol })); } } // 4. Place a Trade Contract (Buy Rise/Fall or Higher/Lower) placeTrade(amount, contractType, currency = 'USD', duration = 1, durationUnit = 'm', symbol = 'R_100') { if (!this.isAuthorized) { console.error("You must authorize your API token before placing a trade."); return; } const proposalRequest = { buy: 1, price: amount, parameters: { amount: amount, basis: "stake", contract_type: contractType, // 'CALL' (Rise/Up) or 'PUT' (Fall/Down) currency: currency, duration: duration, duration_unit: durationUnit, symbol: symbol } }; this.socket.send(JSON.stringify(proposalRequest)); } // 5. Route Incoming API Responses to UI Components handleIncomingData(data) { const msgType = data.msg_type; switch (msgType) { case 'authorize': if (data.error) { console.error("Authorization Failed:", data.error.message); this.isAuthorized = false; } else { console.log(`Authorized successfully! Welcome ${data.authorize.email}`); this.isAuthorized = true; // Update UI with Virtual Account Balance ($10,000) document.getElementById('balance-display').innerText = `$${data.authorize.balance}`; } break; case 'tick': if (data.tick) { const price = data.tick.quote; const epoch = data.tick.epoch; this.updateUIChart(price, epoch); } break; case 'buy': if (data.error) { console.error("Trade Execution Failed:", data.error.message); } else { console.log(`Trade placed! Contract ID: ${data.buy.contract_id}`); } break; default: break; } } // 6. UI Render Placeholders (To link with your cloned frontend template) updateUIChart(price, time) { // Send this live stream data straight to your chart UI component const priceElement = document.getElementById('live-price-ticker'); if (priceElement) { priceElement.innerText = price; } } startKeepAlive() { setInterval(() => { if (this.socket.readyState === WebSocket.OPEN) { this.socket.send(JSON.stringify({ ping: 1 })); } }, 30000); } } // Example Usage on your cloned page: // const botEngine = new DerivTradingEngine('YOUR_DERIV_APP_ID'); // botEngine.init(); // botEngine.authorize('YOUR_USER_API_TOKEN'); // botEngine.subscribeToTicks('R_100');
